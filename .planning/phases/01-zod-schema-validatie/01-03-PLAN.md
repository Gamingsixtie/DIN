export const KiBExportSchema = z.object({
  visie: z.object({
    uitgebreid: z.string(),
    beknopt: z.string(),

  doelen: z.array(z.object({

         let parsed: unknown;
         try {
           parsed = JSON.parse(json);
         } catch {
           throw new Error("Ongeldig JSON formaat. Controleer de KiB export.");
         }
         const result = KiBExportSchema.safeParse(parsed);
         if (!result.success) {
           throw new Error(
