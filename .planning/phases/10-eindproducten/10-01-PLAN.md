    ```typescript
    function numberedHeading(text: string, level: 'h1' | 'h2', state: NumberingState): Paragraph {
      if (level === 'h1') {
        state.h1Counter++;
        state.h2Counter = 0;
        return heading(`${state.h1Counter}. ${text}`, HeadingLevel.HEADING_1);
      } else {
        state.h2Counter++;
        return heading(`${state.h1Counter}.${state.h2Counter} ${text}`, HeadingLevel.HEADING_2);
      }
    }
    ```

    5. Create exported `categorizeGaps()` function:
    ```typescript
    export function categorizeGaps(session: DINSession) {
      const gaps = findGaps(
        session.goals, session.benefits, session.capabilities, session.efforts,
        session.goalBenefitMaps, session.benefitCapabilityMaps, session.capabilityEffortMaps
      );
      const volgendeCyclus: typeof session.goals = [];
      const echteGapsGoals: typeof session.goals = [];
      gaps.goalsWithoutBenefits.forEach(id => {
        const goal = session.goals.find(g => g.id === id);
        if (!goal) return;
        const status = getGoalCompletionStatus(session, id);
        if (status.status === "niet-begonnen") {
          volgendeCyclus.push(goal);
        } else {
          echteGapsGoals.push(goal);
        }
      });
      return {
        volgendeCyclus,
        echteGapsGoals,
        benefitsWithoutCaps: gaps.benefitsWithoutCapabilities
          .map(id => session.benefits.find(b => b.id === id))
          .filter(Boolean),
        capsWithoutEfforts: gaps.capabilitiesWithoutEfforts
          .map(id => session.capabilities.find(c => c.id === id))
          .filter(Boolean),
      };
    }
    ```
