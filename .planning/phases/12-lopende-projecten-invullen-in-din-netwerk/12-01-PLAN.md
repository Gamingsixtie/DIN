    }

      projectId: string,
      maps: ProjectCapabilityMap[],
      capabilities: DINCapability[]
    ): DINCapability[] {
      const capIds = maps.filter(m => m.projectId === projectId).map(m => m.capabilityId);
      return capabilities.filter(c => capIds.includes(c.id));
